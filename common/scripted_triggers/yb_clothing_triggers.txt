			}
			OR = {
				regional_portrait_trigger = { REGION = special_ottoman_conquest_region_1 }
				regional_portrait_trigger = { REGION = special_ottoman_conquest_region_2 }
				regional_portrait_trigger = { REGION = special_ottoman_conquest_region_3 }
				trigger_if = {
					limit = { current_year >= 1500 }
					regional_portrait_trigger = { REGION = special_ottoman_conquest_region_4 }
					regional_portrait_trigger = { REGION = special_ottoman_conquest_region_5 }
				}
				regional_portrait_trigger = { REGION = ghw_region_anatolia }
				regional_portrait_trigger = { REGION = ghw_region_greece }
				regional_portrait_trigger = { REGION = ghw_region_thessalonika }
			}
		}
		portrait_shared_clothing_contents_trigger = {
			CULTURE_SCOPE = scope:culture
			CULTURE_FLAG = epe_seljuk
		}
	}
	trigger_else = {
		portrait_shared_clothing_contents_trigger = {
			CULTURE_SCOPE = scope:culture
			CULTURE_FLAG = yb_ottoman
		}
	}

	portrait_era4_trigger = yes
	trigger_if = {
		limit = { exists = this	}
		NOR = {
			government_has_flag = government_has_herd
			government_has_flag = government_has_merit
		}
	}

regional_portrait_trigger = {
	OR = {
		trigger_if = {
			limit = { exists = capital_province }
			capital_province = { geographical_region = $REGION$ }
		}
		trigger_else_if = {
			limit = { exists = primary_title.title_capital_county.title_province }
			primary_title.title_capital_county.title_province = { geographical_region = $REGION$ }
		}

		trigger_if = {
			limit = { exists = host.capital_province }
			host.capital_province = { geographical_region = $REGION$ }
		}
		trigger_else_if = {
			limit = { exists = host.primary_title.title_capital_county.title_province }
			host.primary_title.title_capital_county.title_province = { geographical_region = $REGION$ }
		}

		trigger_if = {
			limit = { exists = liege.capital_province }
			liege.capital_province = { geographical_region = $REGION$ }
		}
		trigger_else_if = {
			limit = { exists = liege.primary_title.title_capital_county.title_province }
			liege.primary_title.title_capital_county.title_province = { geographical_region = $REGION$ }
		}

		trigger_if = {
			limit = { exists = top_liege.capital_province }
			top_liege.capital_province = { geographical_region = $REGION$ }
		}
		trigger_else_if = {
			limit = { exists = top_liege.primary_title.title_capital_county.title_province }
			top_liege.primary_title.title_capital_county.title_province = { geographical_region = $REGION$ }
		}

		trigger_if = {
			limit = { exists = location.province_owner.capital_province }
			location.province_owner.capital_province = { geographical_region = $REGION$ }
		}
		trigger_else_if = {
			limit = { exists = location.province_owner.primary_title.title_capital_county.title_province }
			location.province_owner.primary_title.title_capital_county.title_province = { geographical_region = $REGION$ }
		}
	}
}

rooted_religion_portrait_trigger = {
	root.religion = religion:$ROOTED_RELIGION$
	#trigger_if = {
	#	limit = { exists = root.primary_title.previous_holder }
	#	root.primary_title.previous_holder.religion = religion:$ROOTED_RELIGION$
	#}
	trigger_if = { #trigger_else_if
		limit = {
			exists = root.father
			root.father = { is_alive = yes }
			NOT = { has_trait = disinherited }
		}
		root.father.religion = religion:$ROOTED_RELIGION$
	}
	trigger_else_if = {
		limit = {
			exists = capital_province
			age < 18
		}
		root.capital_province.faith.religion = religion:$ROOTED_RELIGION$
	}
	trigger_else_if = {
		limit = { exists = root.top_liege }
		root.top_liege.religion = religion:$ROOTED_RELIGION$
	}
	trigger_else_if = {
		limit = { exists = root.liege }
		root.liege.religion = religion:$ROOTED_RELIGION$
	}
	trigger_else_if = {
		limit = { exists = root.host }
		root.host.religion = religion:$ROOTED_RELIGION$
	}
	trigger_else_if = {
		limit = {
			exists = root.mother
			root.mother = { is_alive = yes }
			OR = {
				root = { is_ruler = no }
				root.mother = { is_spouse_of = root.father }
				root.mother.top_liege = root
				root.mother.liege = root
				root.mother.host = root
			}
		}
		root.mother.religion = religion:$ROOTED_RELIGION$
	}
}

		}

			}
		}
		AND = {

			portrait_wear_armor_trigger = yes
			OR = {

					OR = {
						using_cb = undirected_great_holy_war

					}
					is_attacker = prev
				}
				any_liege_or_above = {
